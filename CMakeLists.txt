cmake_minimum_required(VERSION 3.20)
project(srt_multi_protocol_support VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
option(SRTMUX_ENABLE_SANITIZERS "Enable address/undefined sanitizers on supported compilers" OFF)

add_library(srtmux_host STATIC src/module.cpp src/transport.cpp)
add_library(SrtMux::host ALIAS srtmux_host)
target_compile_features(srtmux_host PUBLIC cxx_std_20)
target_include_directories(srtmux_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(srtmux_host PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(srtmux_host PRIVATE ws2_32)
endif()

function(srtmux_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(SRTMUX_ENABLE_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "This sanitizer configuration requires GCC or Clang")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
srtmux_warnings(srtmux_host)

option(SRTMUX_BUILD_PROVIDERS "Build pinned real SRT provider modules (downloads sources)" OFF)
if(SRTMUX_BUILD_PROVIDERS)
  add_subdirectory(providers)
endif()

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
