set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)

set(ENABLE_UNITTESTS OFF CACHE BOOL "" FORCE)

set(robotweax_revision 29c741d35f89a412926c374d6ccb15fa1270833c)
set(haivision_revision 899348d8318eb9a3c5a5b6ec43c4a1114288773a)
FetchContent_Declare(robotweax
  URL https://codeload.github.com/Robotweax/srt/tar.gz/${robotweax_revision}
  URL_HASH SHA256=325d8ecf79b7798faff833499ebfa6b63ea49d83bebb10e2a23aaca26b3f990b)
FetchContent_Declare(haivision
  URL https://codeload.github.com/Haivision/srt/tar.gz/${haivision_revision}
  URL_HASH SHA256=82050d96a1a55d54b423fed4ad80ddb6f30b31bf33e317ffd6726c91ee9a6170)
if(SRTMUX_ENABLE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
FetchContent_MakeAvailable(robotweax haivision)
include("${CMAKE_CURRENT_SOURCE_DIR}/haivision_close_patch.cmake")
set_target_properties(srt_virtual PROPERTIES POSITION_INDEPENDENT_CODE ON)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/licenses")
configure_file("${robotweax_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/licenses/Robotweax-LICENSE" COPYONLY)
configure_file("${haivision_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/licenses/Haivision-LICENSE" COPYONLY)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/adapter.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/polling.hpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/adapter.cpp" adapter_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/polling.hpp" polling_hash)
foreach(provider IN ITEMS robotweax haivision)
  add_library(srtmux_${provider} MODULE adapter.cpp)
  target_compile_features(srtmux_${provider} PRIVATE cxx_std_20)
  target_include_directories(srtmux_${provider} PRIVATE "${PROJECT_SOURCE_DIR}/include")
  if(provider STREQUAL "robotweax")
    set(features "static-srt,shared-openssl,basic-io,native-poller")
    set(release 0.3.3)
    set(tag 0x5257)
    target_include_directories(srtmux_${provider} PRIVATE "${robotweax_SOURCE_DIR}/include/srt")
    target_link_libraries(srtmux_${provider} PRIVATE robotweax_srt)
  else()
    set(features "static-srt,shared-openssl,basic-io,native-poller,km-close-fix=${haivision_patch_hash}")
    set(release 1.5.7)
    set(tag 0x4856)
    target_compile_definitions(srtmux_${provider} PRIVATE PROVIDER_HAIVISION=1)
    target_include_directories(srtmux_${provider} PRIVATE "${haivision_SOURCE_DIR}/srtcore" "${haivision_BINARY_DIR}")
    target_link_libraries(srtmux_${provider} PRIVATE srt_static)
  endif()
  string(SHA256 fingerprint "${provider};${${provider}_revision};${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_SYSTEM_NAME};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_BUILD_TYPE};${CMAKE_CXX_FLAGS};${OPENSSL_VERSION};${USE_ENCLIB};${SRTMUX_ENABLE_SANITIZERS};${adapter_hash};${polling_hash};${features};adapter-polling-v1")
  set(fingerprint_${provider} "${fingerprint}")
  target_compile_definitions(srtmux_${provider} PRIVATE SRTM_BUILD_PROVIDER
    PROVIDER_NAME="${provider}" PROVIDER_RELEASE="${release}" PROVIDER_TAG=${tag}
    PROVIDER_REVISION="${${provider}_revision}" PROVIDER_FINGERPRINT="${fingerprint}"
    PROVIDER_PLATFORM="${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}"
    PROVIDER_BUILD_FEATURES="${features}")
  set_target_properties(srtmux_${provider} PROPERTIES CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES WINDOWS_EXPORT_ALL_SYMBOLS OFF)
  if(APPLE)
    target_link_options(srtmux_${provider} PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.macos")
  elseif(UNIX)
    target_link_options(srtmux_${provider} PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.elf" "LINKER:-Bsymbolic")
  endif()
  srtmux_warnings(srtmux_${provider})
endforeach()

option(SRTMUX_BUILD_BENCHMARKS "Build direct-versus-module File/IPv4 benchmark" OFF)
if(SRTMUX_BUILD_BENCHMARKS)
  find_package(Threads REQUIRED)
  add_executable(srtmux_facade_benchmark ../benchmarks/throughput.cpp)
  target_link_libraries(srtmux_facade_benchmark PRIVATE SrtMux::host Threads::Threads)
  target_compile_definitions(srtmux_facade_benchmark PRIVATE SRTMUX_FACADE=1)
  srtmux_warnings(srtmux_facade_benchmark)
  add_executable(srtmux_module_benchmark ../benchmarks/throughput.cpp)
  target_link_libraries(srtmux_module_benchmark PRIVATE SrtMux::host Threads::Threads)
  srtmux_warnings(srtmux_module_benchmark)
  foreach(provider IN ITEMS robotweax haivision)
    add_executable(srtmux_direct_${provider} ../benchmarks/throughput.cpp)
    target_compile_features(srtmux_direct_${provider} PRIVATE cxx_std_20)
    target_include_directories(srtmux_direct_${provider} PRIVATE "${PROJECT_SOURCE_DIR}/include")
    target_compile_definitions(srtmux_direct_${provider} PRIVATE SRTMUX_DIRECT=1 BENCHMARK_PROVIDER="${provider}" BENCHMARK_FINGERPRINT="${fingerprint_${provider}}")
    if(provider STREQUAL "robotweax")
      target_include_directories(srtmux_direct_${provider} PRIVATE "${robotweax_SOURCE_DIR}/include/srt")
      target_link_libraries(srtmux_direct_${provider} PRIVATE robotweax_srt Threads::Threads)
    else()
      target_include_directories(srtmux_direct_${provider} PRIVATE "${haivision_SOURCE_DIR}/srtcore" "${haivision_BINARY_DIR}")
      target_link_libraries(srtmux_direct_${provider} PRIVATE srt_static Threads::Threads)
    endif()
    srtmux_warnings(srtmux_direct_${provider})
  endforeach()
endif()
